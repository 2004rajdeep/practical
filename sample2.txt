import numpy as np

def fuzzy_union_or(A, B):
    if len(A) != len(B):
        raise ValueError("Fuzzy sets must have the same length.")
    return np.maximum(A, B)

def fuzzy_intersection_and(A, B):
    if len(A) != len(B):
        raise ValueError("Fuzzy sets must have the same length.")
    return np.minimum(A, B)

def fuzzy_complement_not(A):
    return 1 - A

import numpy as np
import matplotlib.pyplot as plt

# --- Step Function (Activation) ---
def step_function(weighted_sum):
    return 1 if weighted_sum >= 0 else 0

# --- Perceptron Class ---
class Perceptron:
    def __init__(self, num_inputs, learning_rate=0.1, max_epochs=50):
        self.weights       = np.random.uniform(-0.5, 0.5, num_inputs)
        self.bias          = np.random.uniform(-0.5, 0.5)
        self.learning_rate = learning_rate
        self.max_epochs    = max_epochs
        self.errors        = []

    def predict(self, inputs):
        weighted_sum = np.dot(inputs, self.weights) + self.bias
        return step_function(weighted_sum)

    def train(self, X_train, y_train):
        print(f"--- Training Perceptron (LR={self.learning_rate}, Epochs={self.max_epochs}) ---\n")

        for epoch in range(self.max_epochs):
            total_error = 0

            for inputs, label in zip(X_train, y_train):
                prediction  = self.predict(inputs)
                error       = label - prediction
                total_error += abs(error)

                if error != 0:
                    # Update weights and bias
                    self.weights += self.learning_rate * error * inputs
                    self.bias    += self.learning_rate * error

            self.errors.append(total_error)

            # Stop early if fully converged
            if total_error == 0:
                print(f"✅ Converged at Epoch {epoch + 1}!")
                break

            if (epoch + 1) % 10 == 0:
                print(f"Epoch {epoch+1} | Total Errors: {total_error}")

        print(f"\nFinal Weights : {self.weights}")
        print(f"Final Bias    : {self.bias:.4f}")

y_train = np.array([0, 0, 0, 1])

import numpy as np
import matplotlib.pyplot as plt

# --- Activation Function ---
def sigmoid(x):
    x = np.clip(x, -500, 500)       # Prevent overflow
    return 1 / (1 + np.exp(-x))

def sigmoid_derivative(output):
    return output * (1 - output)    # Derivative using output value

# --- MLP Class ---
class MLP:
    def __init__(self, input_size, hidden_size, output_size, lr=0.2, epochs=10000):
        self.lr      = lr
        self.epochs  = epochs
        self.errors  = []

        # Random weight initialization
        self.W_ih = np.random.uniform(-0.5, 0.5, (input_size, hidden_size))  # Input→Hidden
        self.b_h  = np.zeros((1, hidden_size))                                # Hidden bias

        self.W_ho = np.random.uniform(-0.5, 0.5, (hidden_size, output_size)) # Hidden→Output
        self.b_o  = np.zeros((1, output_size))                                # Output bias

    def forward(self, X):
        # Hidden layer
        self.net_h = np.dot(X, self.W_ih) + self.b_h
        self.out_h = sigmoid(self.net_h)

        # Output layer
        self.net_o = np.dot(self.out_h, self.W_ho) + self.b_o
        self.out_o = sigmoid(self.net_o)
        return self.out_o

    def backward(self, X, y):
        # --- Output layer error ---
        error_o = y - self.out_o
        delta_o = error_o * sigmoid_derivative(self.out_o)

        # --- Hidden layer error (backpropagate) ---
        error_h = delta_o.dot(self.W_ho.T)
        delta_h = error_h * sigmoid_derivative(self.out_h)

        # --- Update weights ---
        self.W_ho += self.out_h.T.dot(delta_o) * self.lr
        self.b_o  += np.sum(delta_o, axis=0, keepdims=True) * self.lr

        self.W_ih += X.T.dot(delta_h) * self.lr
        self.b_h  += np.sum(delta_h, axis=0, keepdims=True) * self.lr

        return np.mean(error_o ** 2)   # Return MSE

    def train(self, X, y):
        print(f"--- Training MLP (Hidden={self.W_ih.shape[1]}, LR={self.lr}) ---\n")
        for epoch in range(self.epochs):
            out = self.forward(X)
            mse = self.backward(X, y)
            self.errors.append(mse)

            if (epoch + 1) % 1000 == 0:
                print(f"Epoch {epoch+1}/{self.epochs} | MSE: {mse:.6f}")
        print("\n--- Training Complete ---")

# --- XOR Gate Data ---
X_train = np.array([[0,0], [0,1], [1,0], [1,1]])
y_train = np.array([[0],   [1],   [1],   [0]])

# --- Train MLP ---
mlp = MLP(input_size=2, hidden_size=4, output_size=1, lr=0.2, epochs=10000)
mlp.train(X_train, y_train)

# --- Test ---
print("\n--- Testing Model ---")
predictions = mlp.forward(X_train)
for inputs, pred, expected in zip(X_train, predictions, y_train):

plt.figure(figsize=(10, 5))
plt.plot(mlp.errors, color='red')
plt.title('MLP Training Error (MSE) Over Epochs')
plt.xlabel('Epoch')
plt.ylabel('Mean Squared Error')
plt.grid(True)

# --- Threshold Activation Function ---
def mcp_activation(net_input, threshold):
    return 1 if net_input >= threshold else 0

# --- MCP Neuron ---
def mcp_neuron(inputs, weights, threshold):
    inputs    = np.array(inputs)
    weights   = np.array(weights)
    net_input = np.dot(inputs, weights)   # Weighted sum
    output    = mcp_activation(net_input, threshold)
    return net_input, output

# Practical 7A: Implementation of Genetic Algorithm (GA)
# Topic: Evolutionary Computing
# Goal: Find maximum of f(x) = x * sin(10*pi*x) + 1 in range [-1, 2]
# Operations: Tournament Selection, Arithmetic Crossover, Gaussian Mutation
# Library: numpy, matplotlib
# How to run: python practical7a.py

import numpy as np
import matplotlib.pyplot as plt

# Fitness function to maximize
def fitness_function(x):
    return x * np.sin(10 * np.pi * x) + 1

# GA Parameters
POP_SIZE       = 40
GENERATIONS    = 100
X_MIN, X_MAX   = -1.0, 2.0
CROSSOVER_PROB = 0.9
MUTATION_PROB  = 0.2
MUTATION_STD   = 0.1

np.random.seed(42)
population = np.random.uniform(X_MIN, X_MAX, POP_SIZE)

# Tournament Selection - pick best from random k candidates
def tournament_selection(pop, fitness_vals, k=3):
    selected = []
    for _ in range(len(pop)):
        idx = np.random.choice(len(pop), k, replace=False)
        selected.append(pop[idx[np.argmax(fitness_vals[idx])]])
    return np.array(selected)

# Arithmetic Crossover - blend two parents
def arithmetic_crossover(p1, p2):
    alpha = np.random.rand()
    return alpha * p1 + (1 - alpha) * p2, alpha * p2 + (1 - alpha) * p1

# Gaussian Mutation - add small random noise
def mutate(x):
    if np.random.rand() < MUTATION_PROB:
        x += np.random.normal(0, MUTATION_STD)
    return np.clip(x, X_MIN, X_MAX)

# GA Main Loop
best_history = []
mean_history = []

for _ in range(GENERATIONS):
    fitness_vals = fitness_function(population)
    best_history.append(np.max(fitness_vals))
    mean_history.append(np.mean(fitness_vals))

    parents   = tournament_selection(population, fitness_vals)
    offspring = []
    np.random.shuffle(parents)

    for i in range(0, POP_SIZE, 2):
        if np.random.rand() < CROSSOVER_PROB:
            c1, c2 = arithmetic_crossover(parents[i], parents[i+1])
        else:
            c1, c2 = parents[i], parents[i+1]
        offspring.extend([mutate(c1), mutate(c2)])

    population = np.array(offspring)

# Results
best_x   = population[np.argmax(fitness_function(population))]
print(f"Best x found    : {best_x:.4f}")
print(f"Best fitness    : {fitness_function(best_x):.4f}")

# Plot 1: Convergence
plt.figure(figsize=(10, 4))
plt.plot(best_history, label='Best Fitness')
plt.plot(mean_history, label='Mean Fitness')
plt.title('GA Convergence Over Generations')
plt.xlabel('Generation')
plt.ylabel('Fitness')
plt.legend()
plt.grid(True)
plt.show()

# Plot 2: Final Population on Function
x = np.linspace(X_MIN, X_MAX, 500)
plt.figure(figsize=(10, 4))
plt.plot(x, fitness_function(x), label='f(x)', color='blue')
plt.scatter(population, fitness_function(population), color='red', label='Final Population')
plt.title('Final Population on Fitness Function')
plt.xlabel('x')
plt.ylabel('f(x)')
plt.legend()
plt.grid(True)

# Practical 7B: Genetic Algorithm - Minimize f(x) = x^2
# Topic: Evolutionary Computing
# Goal: Find x closest to 0 where f(x) = x^2 is minimum
# Operations: Tournament Selection, Arithmetic Crossover, Random Mutation
# Library: random only
# How to run: python practical7b.py

import random

def fitness(x):
    return x * x                          # Minimize x^2 (min at x=0)

def initialize_population(size):
    return [random.uniform(-10, 10) for _ in range(size)]

def selection(population):
    a, b = random.sample(population, 2)
    return a if fitness(a) < fitness(b) else b   # Lower fitness = better

def crossover(p1, p2):
    alpha = random.random()
    return alpha * p1 + (1 - alpha) * p2

def mutation(child, rate=0.1):
    if random.random() < rate:
        child += random.uniform(-1, 1)
    return child

def genetic_algorithm():
    pop_size    = 20
    generations = 50
    population  = initialize_population(pop_size)

    for gen in range(generations):
        new_population = []
        for _ in range(pop_size):
            p1    = selection(population)
            p2    = selection(population)
            child = crossover(p1, p2)
            child = mutation(child)
            new_population.append(child)
        population = new_population

    best = min(population, key=fitness)
    print(f"Best solution (x) : {best:.6f}")
    print(f"Minimum f(x) = x² : {fitness(best):.6f}")

genetic_algorithm()

# Practical 8A: Solving Travelling Salesman Problem (TSP) using GA
# Topic: Combinatorial Optimization with Genetic Algorithm
# Goal: Find shortest route visiting all 4 cities exactly once
# Operations: Tournament Selection, Order Crossover (OX), Swap Mutation
# Library: numpy, random
# How to run: python practical8a.py

import random
import numpy as np

# Distance matrix between 4 cities
distance = np.array([
    [0, 10, 15, 20],
    [10, 0, 35, 25],
    [15, 35, 0, 30],
    [20, 25, 30, 0]
])

NUM_CITIES    = 4
POP_SIZE      = 10
GENERATIONS   = 100
MUTATION_RATE = 0.2

def total_distance(tour):
    return sum(distance[tour[i], tour[(i+1) % NUM_CITIES]]
               for i in range(NUM_CITIES))

def fitness(tour):
    return 1 / total_distance(tour)      # Higher = shorter route

def create_population():
    return [random.sample(range(NUM_CITIES), NUM_CITIES)
            for _ in range(POP_SIZE)]

def selection(population):
    a, b = random.sample(population, 2)
    return a if fitness(a) > fitness(b) else b

def crossover(p1, p2):
    start, end = sorted(random.sample(range(NUM_CITIES), 2))
    child      = [-1] * NUM_CITIES
    child[start:end] = p1[start:end]
    pointer = end
    for city in p2:
        if city not in child:
            if pointer >= NUM_CITIES:
                pointer = 0
            child[pointer] = city
            pointer += 1
    return child

def mutation(tour):
    if random.random() < MUTATION_RATE:
        i, j         = random.sample(range(NUM_CITIES), 2)
        tour[i], tour[j] = tour[j], tour[i]
    return tour

# GA Main Loop
population = create_population()
for _ in range(GENERATIONS):
    new_population = []
    for _ in range(POP_SIZE):
        p1    = selection(population)
        p2    = selection(population)
        child = crossover(p1, p2)
        child = mutation(child)
        new_population.append(child)
    population = new_population

best_tour = min(population, key=total_distance)
print(f"Best Tour     : {best_tour}")
print(f"Min Distance  : {total_distance(best_tour)}")

def mutate(route):
    if random.random() < MUTATION_RATE:
        i, j = random.sample(range(NUM_CITIES), 2)
        route[i], route[j] = route[j], route[i]
    return route

population = init_population()
best_dist  = []

for _ in range(GENERATIONS):
    new_pop = []
    for _ in range(POP_SIZE):
        p1    = selection(population)
        p2    = selection(population)
        child = crossover(p1, p2)
        child = mutate(child)
        new_pop.append(child)
    population   = new_pop
    best_route   = min(population, key=total_distance)
    best_dist.append(total_distance(best_route))

print(f"Best Route       : {best_route}")
print(f"Minimum Distance : {total_distance(best_route)}")

plt.figure(figsize=(10, 5))
plt.plot(best_dist, color='green')
plt.title('GA for TSP - Convergence')
plt.xlabel('Generation')
plt.ylabel('Best Distance')
plt.grid(True)
plt.show()

# Practical 11: Particle Swarm Optimization (PSO)
# Topic: Swarm Intelligence
# Goal: Find global minimum of Rastrigin function f(x) = 0 at x=[0,0]
# Parameters: w=inertia, c1=cognitive, c2=social
# Library: numpy
# How to run: python practical11.py

import numpy as np

def rastrigin(X):
    A = 10
    return A * len(X) + np.sum(X**2 - A * np.cos(2 * np.pi * X))

# PSO Parameters
num_particles = 30
dimensions    = 2
iterations    = 100
w, c1, c2     = 0.7, 1.5, 1.5

# Initialize particles randomly
X = np.random.uniform(-5.12, 5.12, (num_particles, dimensions))
V = np.random.uniform(-1, 1,       (num_particles, dimensions))

pbest     = X.copy()
pbest_val = np.array([rastrigin(x) for x in X])
gbest     = pbest[np.argmin(pbest_val)]
gbest_val = min(pbest_val)

# PSO Main Loop
for _ in range(iterations):
    for i in range(num_particles):
        r1, r2 = np.random.rand(), np.random.rand()
        V[i]   = w * V[i] + c1 * r1 * (pbest[i] - X[i]) + c2 * r2 * (gbest - X[i])
        X[i]   = X[i] + V[i]

        fit = rastrigin(X[i])
        if fit < pbest_val[i]:
            pbest[i]     = X[i].copy()
            pbest_val[i] = fit

    gbest     = pbest[np.argmin(pbest_val)]
    gbest_val = min(pbest_val)

print(f"Global Minimum Position : {gbest}")
print(f"Minimum Value           : {gbest_val:.6f}")

# Practical 12: PSO for Load Balancing (Makespan Minimization)
# Topic: Swarm Intelligence applied to Scheduling
# Goal: Assign 6 tasks to 3 machines to minimize total completion time
# Library: numpy, random
# How to run: python practical12.py

import numpy as np
import random

N         = 6
M         = 3
task_time = np.array([2, 4, 6, 3, 5, 7])

num_particles = 20
iterations    = 50
w, c1, c2     = 0.7, 1.5, 1.5

def fitness(position):
    loads = np.zeros(M)
    for i in range(N):
        machine = int(round(position[i])) % M
        loads[machine] += task_time[i]
    return max(loads)                        # Minimize the max load (makespan)

particles  = np.random.randint(0, M, (num_particles, N)).astype(float)
velocities = np.zeros((num_particles, N))

pbest         = particles.copy()
pbest_fitness = np.array([fitness(p) for p in particles])
gbest         = pbest[np.argmin(pbest_fitness)].copy()
gbest_fitness = min(pbest_fitness)

for _ in range(iterations):
    for i in range(num_particles):
        r1, r2          = random.random(), random.random()
        velocities[i]   = (w * velocities[i]
                         + c1 * r1 * (pbest[i] - particles[i])
                         + c2 * r2 * (gbest   - particles[i]))
        particles[i]    = particles[i] + velocities[i]

        fit = fitness(particles[i])
        if fit < pbest_fitness[i]:
            pbest[i]         = particles[i].copy()
            pbest_fitness[i] = fit

    gbest         = pbest[np.argmin(pbest_fitness)].copy()
    gbest_fitness = min(pbest_fitness)

allocation = np.round(gbest).astype(int) % M
print(f"Task → Machine Allocation : {allocation}")
print(f"Minimum Makespan          : {gbest_fitness}")

# Practical 14: LSTM Sentiment Classification
# Topic: Deep Learning - NLP with LSTM
# Goal: Classify sentences as Positive (1) or Negative (0)
# Library: PyTorch (torch)
# Install: pip install torch
# How to run: python practical14.py

import torch
import torch.nn as nn
import torch.optim as optim

# Synthetic dataset
sentences = [
    "I love this movie",   "This film is great",  "Amazing experience",
    "I hate this movie",   "This film is terrible","Worst experience"
]
labels = [1, 1, 1, 0, 0, 0]

# Tokenization - assign number to each word
vocab = {}
for s in sentences:
    for w in s.lower().split():
        if w not in vocab:
            vocab[w] = len(vocab) + 1

def encode(sentence):
    return [vocab.get(w, 0) for w in sentence.lower().split()]

X       = [encode(s) for s in sentences]
y       = torch.tensor(labels, dtype=torch.float)
max_len = max(len(seq) for seq in X)

# Padding to equal length
X_pad    = [seq + [0] * (max_len - len(seq)) for seq in X]
X_tensor = torch.tensor(X_pad)

# LSTM Model
class LSTMClassifier(nn.Module):
    def __init__(self, vocab_size):
        super().__init__()
        self.embed   = nn.Embedding(vocab_size + 1, 8, padding_idx=0)
        self.lstm    = nn.LSTM(8, 16, batch_first=True)
        self.fc      = nn.Linear(16, 1)
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        x       = self.embed(x)
        _, (h,_)= self.lstm(x)
        return self.sigmoid(self.fc(h[-1]))

model     = LSTMClassifier(len(vocab))
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=0.01)

# Training
for epoch in range(200):
    optimizer.zero_grad()
    output = model(X_tensor).squeeze()
    loss   = criterion(output, y)
    loss.backward()
    optimizer.step()

print("--- Training Complete ---\n")

# Test on new sentence
test_sentence  = "I love this film"
test_encoded   = encode(test_sentence)
test_encoded  += [0] * (max_len - len(test_encoded))
test_tensor    = torch.tensor([test_encoded])
prediction     = model(test_tensor).item()

print(f"Test Sentence : '{test_sentence}'")
print(f"Sentiment Score (0=negative, 1=positive) : {prediction:.4f}")
print(f"Prediction    : {'Positive 😊' if prediction >= 0.5 else 'Negative 😞'}")

# Practical 15: Self-Organizing Map (SOM) - Iris Dataset Clustering
# Topic: Unsupervised Learning - Competitive Neural Networks
# Goal: Cluster Iris dataset using SOM (manual numpy implementation)
# Library: numpy, matplotlib, sklearn (for dataset only)
# How to run: python practical15.py

import numpy as np
import matplotlib.pyplot as plt
from sklearn.datasets import load_iris
from sklearn.preprocessing import MinMaxScaler

# Load and normalize data
iris    = load_iris()
X       = MinMaxScaler().fit_transform(iris.data)

# SOM Parameters
grid_size   = 5           # 5x5 grid of neurons
input_size  = 4           # 4 features in Iris
lr          = 0.5         # Initial learning rate
epochs      = 100
sigma       = 2.0         # Initial neighbourhood radius

# Initialize SOM weights randomly
weights = np.random.rand(grid_size, grid_size, input_size)

def find_bmu(weights, x):
    """Find Best Matching Unit (closest neuron to input x)"""
    distances = np.linalg.norm(weights - x, axis=2)
    return np.unravel_index(np.argmin(distances), distances.shape)

def update_weights(weights, x, bmu, lr, sigma):
    """Update all neurons based on distance from BMU"""
    for i in range(grid_size):
        for j in range(grid_size):
            dist_to_bmu = np.sqrt((i - bmu[0])**2 + (j - bmu[1])**2)
            influence   = np.exp(-dist_to_bmu**2 / (2 * sigma**2))
            weights[i,j] += lr * influence * (x - weights[i,j])
    return weights

# Training
for epoch in range(epochs):
    current_lr    = lr    * np.exp(-epoch / epochs)
    current_sigma = sigma * np.exp(-epoch / epochs)
    np.random.shuffle(X)
    for x in X:
        bmu     = find_bmu(weights, x)
        weights = update_weights(weights, x, bmu, current_lr, current_sigma)

# Map each sample to its BMU
bmu_map = np.zeros((len(X), 2), dtype=int)
for idx, x in enumerate(iris.data):
    x_norm      = MinMaxScaler().fit_transform(iris.data)[idx]
    bmu         = find_bmu(weights, x_norm)
    bmu_map[idx]= bmu

# Plot
plt.figure(figsize=(8, 8))
colors = ['red', 'green', 'blue']
labels = iris.target_names
for i, (bmu, label) in enumerate(zip(bmu_map, iris.target)):
    plt.scatter(bmu[0], bmu[1], color=colors[label], alpha=0.6)

for i, color in enumerate(colors):
    plt.scatter([], [], color=color, label=labels[i])

plt.title('SOM Clustering of Iris Dataset')
plt.xlabel('SOM Grid X')
plt.ylabel('SOM Grid Y')
plt.legend()
plt.grid(True)
plt.show()
